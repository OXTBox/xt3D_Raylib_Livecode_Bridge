-- xt3d diagnostic: runs one frame a step at a time and records each step in
-- C:/xt3d/diag_log.txt BEFORE running it. After a crash, the last line in that
-- file names the call that died.
--
-- Paste into the card script of a NEW stack (not alongside the demo), Apply,
-- then type  xt3dDiag  in the message box.

constant kLog = "C:/xt3d/diag_log.txt"

command xt3dLog pText
   open file kLog for append
   write pText & return to file kLog
   close file kLog
end xt3dLog

on xt3dDiag
   local tTarget, tCamera, tMesh, tModel, tOK, tData
   put empty into URL ("file:" & kLog)

   xt3dLog "01 InitHidden"
   xt3d_InitHidden 640, 480, "xt3d"
   xt3dLog "   window ready: " & xt3d_IsWindowReady()

   xt3dLog "02 LoadRenderTexture"
   put xt3d_LoadRenderTexture(640, 480) into tTarget
   xt3dLog "   handle " & tTarget & " valid: " & xt3d_IsRenderTextureValid(tTarget)

   xt3dLog "03 CameraCreate"
   put xt3d_CameraCreate(6,5,6, 0,0.5,0, 0,1,0, 45, 0) into tCamera
   xt3dLog "   handle " & tCamera & " = " & xt3d_CameraGet(tCamera)

   xt3dLog "04 GenMeshCube"
   put xt3d_GenMeshCube(1, 1, 1) into tMesh
   xt3dLog "   handle " & tMesh & " info " & xt3d_MeshInfo(tMesh)

   xt3dLog "05 LoadModelFromMesh"
   put xt3d_LoadModelFromMesh(tMesh) into tModel
   xt3dLog "   handle " & tModel & " info " & xt3d_ModelInfo(tModel)

   xt3dLog "06 BeginTextureMode"
   xt3d_BeginTextureMode tTarget
   xt3dLog "07 ClearBackground"
   xt3d_ClearBackground 30, 30, 40, 255
   xt3dLog "08 BeginMode3D"
   xt3d_BeginMode3D tCamera
   xt3dLog "09 DrawGrid"
   xt3d_DrawGrid 10, 1
   xt3dLog "10 DrawCube (simple shape)"
   xt3d_DrawCube 2,0.5,0, 1,1,1, 0,200,0,255
   xt3dLog "11 DrawModelEx"
   xt3d_DrawModelEx tModel, 0,0.5,0, 0,1,0, 30, 1,1,1, 255,160,0,255
   xt3dLog "12 EndMode3D"
   xt3d_EndMode3D
   xt3dLog "13 DrawText"
   xt3d_DrawText "xt3d", 10, 10, 20, 255,255,255,255
   xt3dLog "14 EndTextureMode"
   xt3d_EndTextureMode

   xt3dLog "15 RenderTextureExport to C:/xt3d/frame.png"
   put xt3d_RenderTextureExport(tTarget, "C:/xt3d/frame.png") into tOK
   xt3dLog "   export result: " & tOK

   xt3dLog "16 RenderTextureImageData (frame copy into LiveCode)"
   put xt3d_RenderTextureImageData(tTarget) into tData
   xt3dLog "   bytes received: " & the number of bytes in tData & " (expected " & 640 * 480 * 4 & ")"

   xt3dLog "17 set the imageData"
   if there is no image "xt3dView" then create image "xt3dView"
   set the lockLoc of image "xt3dView" to true
   set the width of image "xt3dView" to 640
   set the height of image "xt3dView" to 480
   set the imageData of image "xt3dView" to tData

   xt3dLog "18 finished without crashing"
   answer "Diagnostic finished. See C:/xt3d/diag_log.txt and C:/xt3d/frame.png"
end xt3dDiag
